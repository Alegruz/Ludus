cmake_minimum_required(VERSION 3.29)
project(LudusPersistenceConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Ludus CONFIG REQUIRED COMPONENTS FoundationFilesystem)
add_executable(ludus_persistence_consumer ../persistence.cpp)
target_compile_definitions(ludus_persistence_consumer PRIVATE LUDUS_SDK_PERSISTENCE_STANDALONE=1)
target_link_libraries(ludus_persistence_consumer PRIVATE Ludus::FoundationFilesystem)
ludus_apply_app_policy(ludus_persistence_consumer)
target_compile_options(ludus_persistence_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
