cmake_minimum_required(VERSION 3.29)
project(LudusContentConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Ludus CONFIG REQUIRED COMPONENTS Content)
add_executable(ludus_content_consumer ../content.cpp)
target_link_libraries(ludus_content_consumer PRIVATE Ludus::Content)
target_compile_definitions(ludus_content_consumer PRIVATE LUDUS_SDK_CONTENT_STANDALONE=1)
target_compile_options(ludus_content_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
ludus_apply_app_policy(ludus_content_consumer)
