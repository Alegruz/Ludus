# Tests use only the installed surface and project-owned cooked inputs.
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(cooked "${PROJECT_SOURCE_DIR}/out/script-provider/cooked")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/script-provider" verify
    RESULT_VARIABLE cooked_result OUTPUT_QUIET)
if(NOT cooked_result EQUAL 0)
    message(FATAL_ERROR "Prepare paired tools using scripts/script-provider bootstrap")
endif()
file(READ "${cooked}/current.json" pointer)
string(JSON key GET "${pointer}" key)
set(cooked "${cooked}/${key}")
add_executable(ludus_behavior_acceptance consumer/main.cpp)
target_include_directories(ludus_behavior_acceptance PRIVATE "${cooked}")
target_link_libraries(ludus_behavior_acceptance PRIVATE Ludus::Behavior)
ludus_apply_project_defaults(ludus_behavior_acceptance)
enable_testing()
add_test(NAME ludus_behavior_cook COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_cook.py")
set_tests_properties(ludus_behavior_cook PROPERTIES TIMEOUT 120 LABELS "scripting;behavior;tooling")
if(EMSCRIPTEN)
    target_link_options(ludus_behavior_acceptance PRIVATE -sEXIT_RUNTIME=1 -sENVIRONMENT=web,node
        -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sSTACK_SIZE=1048576 -sSTACK_OVERFLOW_CHECK=2)
endif()
add_test(NAME ludus_behavior_acceptance COMMAND ludus_behavior_acceptance)
set_tests_properties(ludus_behavior_acceptance PROPERTIES TIMEOUT 60 LABELS "scripting;behavior")
if(LUDUS_BUILD_TESTS)
    find_package(Catch2 3 REQUIRED CONFIG)
    add_executable(ludus_behavior_tests tests.cpp)
    target_include_directories(ludus_behavior_tests PRIVATE "${cooked}")
    target_compile_definitions(ludus_behavior_tests PRIVATE "LUDUS_BEHAVIOR_PACKAGE=\"${cooked}/behavior.lupack\"")
    target_link_libraries(ludus_behavior_tests PRIVATE Ludus::Behavior Catch2::Catch2WithMain)
    ludus_apply_project_defaults(ludus_behavior_tests)
    ludus_enable_test_exceptions(ludus_behavior_tests)
    add_test(NAME ludus_behavior_tests COMMAND ludus_behavior_tests)
    set_tests_properties(ludus_behavior_tests PROPERTIES TIMEOUT 60 LABELS "unit;scripting;behavior")
endif()
