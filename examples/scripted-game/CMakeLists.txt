cmake_minimum_required(VERSION 3.29)
project(ludus_scripted_game LANGUAGES CXX)
include(CTest)
find_package(Ludus CONFIG REQUIRED COMPONENTS GameApi GameHost Behavior FoundationParsingJson)
ludus_cook_behaviors(NAME encounter CONTRACT behaviors/contract.json PACKAGE behaviors/package.json)
ludus_add_game(NAME scripted_game SOURCES src/game.cpp)
foreach(target scripted_game_module scripted_game_shipping)
    if(TARGET ${target})
        add_dependencies(${target} encounter_cook)
        target_include_directories(${target} PRIVATE "${encounter_DIRECTORY}")
        target_link_libraries(${target} PRIVATE Ludus::Behavior)
    endif()
endforeach()
if(TARGET scripted_game_module)
    target_link_libraries(scripted_game_module PRIVATE Ludus::FoundationParsingJson)
    # All engine calls belong to linked archives or the explicit services table.
    # Sanitized modules resolve instrumentation through the sanitized host.
    if(NOT Ludus_SDK_ASAN AND NOT Ludus_SDK_UBSAN AND NOT Ludus_SDK_TSAN)
        target_link_options(scripted_game_module PRIVATE "$<$<PLATFORM_ID:Linux>:LINKER:--no-undefined>")
    endif()
endif()
if(BUILD_TESTING)
    add_test(NAME scripted_game_static COMMAND scripted_game_shipping --headless --max-frames 2)
endif()

option(SCRIPTED_GAME_QUALIFICATION "Build the opt-in S7 reference workload" OFF)
if(SCRIPTED_GAME_QUALIFICATION)
    add_subdirectory(benchmarks)
endif()
