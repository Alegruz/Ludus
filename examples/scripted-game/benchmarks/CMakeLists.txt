# Benchmark-only targets inherit the selected SDK's instrumentation as well as
# the public exception-free application policy; they are not Catch2 targets.
function(qualification_policy target)
    ludus_apply_app_policy(${target})
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    foreach(pair ASAN,address UBSAN,undefined TSAN,thread)
        string(REPLACE "," ";" parts "${pair}")
        list(GET parts 0 setting)
        list(GET parts 1 flag)
        if(Ludus_SDK_${setting})
            target_compile_options(${target} PRIVATE -fsanitize=${flag} -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=${flag})
        endif()
    endforeach()
endfunction()
find_package(Ludus CONFIG REQUIRED COMPONENTS FoundationTime)
set(inputs "${CMAKE_CURRENT_BINARY_DIR}/inputs")
file(MAKE_DIRECTORY "${inputs}")
foreach(name encounter.luau encounter.json)
    configure_file("${PROJECT_SOURCE_DIR}/behaviors/${name}" "${inputs}/${name}" COPYONLY)
endforeach()
foreach(name empty.luau scalar.luau operation.luau package.json)
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/${name}" "${inputs}/${name}" COPYONLY)
endforeach()
ludus_cook_behaviors(NAME qualification CONTRACT "${PROJECT_SOURCE_DIR}/behaviors/contract.json"
    PACKAGE "${inputs}/package.json")
foreach(setting SAMPLES WARMUP ACTIVE INACTIVE HEAP SAFEPOINTS)
    if(NOT DEFINED SCRIPTED_GAME_${setting} OR NOT SCRIPTED_GAME_${setting} MATCHES "^[0-9]+$")
        message(FATAL_ERROR "Set SCRIPTED_GAME_${setting} from the S7 workload profile")
    endif()
endforeach()
foreach(target scripted_game_qualification scripted_game_allocations)
    add_executable(${target} main.cpp)
    add_dependencies(${target} qualification_cook)
    target_include_directories(${target} PRIVATE "${qualification_DIRECTORY}")
    target_link_libraries(${target} PRIVATE Ludus::Behavior Ludus::FoundationTime)
    qualification_policy(${target})
    foreach(setting SAMPLES WARMUP ACTIVE INACTIVE HEAP SAFEPOINTS)
        target_compile_definitions(${target} PRIVATE "LUDUS_S7_${setting}=${SCRIPTED_GAME_${setting}}")
    endforeach()
endforeach()
target_sources(scripted_game_allocations PRIVATE probe.cpp)
target_compile_definitions(scripted_game_allocations PRIVATE LUDUS_S7_ALLOCATION_PROBE=1)
add_test(NAME scripted_game_equivalence COMMAND scripted_game_qualification --verify)
add_test(NAME scripted_game_allocations COMMAND scripted_game_allocations --measure)
set_tests_properties(scripted_game_equivalence scripted_game_allocations PROPERTIES TIMEOUT 120)
