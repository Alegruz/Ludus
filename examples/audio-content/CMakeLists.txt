cmake_minimum_required(VERSION 3.29)
project(LudusAudioContentSample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Ludus CONFIG REQUIRED)
add_executable(audio_content_sample main.cpp)
target_link_libraries(audio_content_sample PRIVATE Ludus::AudioContent Ludus::FoundationLogging Ludus::FoundationMath)
ludus_apply_app_policy(audio_content_sample)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME generate_audio COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/generate_fixtures.py" "${CMAKE_CURRENT_BINARY_DIR}/content")
set_tests_properties(generate_audio PROPERTIES FIXTURES_SETUP audio_content)
add_test(NAME audio_content_sample COMMAND audio_content_sample --offline "${CMAKE_CURRENT_BINARY_DIR}/content")
set_tests_properties(audio_content_sample PROPERTIES FIXTURES_REQUIRED audio_content)
