cmake_minimum_required(VERSION 3.29)
project(LudusReferenceScene LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Ludus CONFIG REQUIRED COMPONENTS GraphicsRenderer Resources Content FoundationFilesystem FoundationTime FoundationLogging)
add_executable(reference_scene src/main.cpp)
target_link_libraries(reference_scene PRIVATE Ludus::GraphicsRenderer Ludus::Resources Ludus::Content Ludus::FoundationFilesystem Ludus::FoundationTime Ludus::FoundationLogging)
ludus_apply_app_policy(reference_scene)
foreach(shader renderer_flat renderer_composite renderer_lit renderer_shadow renderer_output)
    ludus_compile_shader(TARGET reference_scene NAME ${shader} RASTER
        SOURCE "${Ludus_DIR}/shaders/${shader}.slang" VERTEX vertexMain FRAGMENT fragmentMain)
endforeach()
ludus_compile_shader(TARGET reference_scene NAME renderer_lit_shadow RASTER
    SOURCE "${Ludus_DIR}/shaders/renderer_lit.slang" VERTEX vertexMain FRAGMENT shadowMain)
ludus_compile_ray_queries(TARGET reference_scene NAME ray_query)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(reference_scene PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2 -Wundef
        -Wnull-dereference -Wdouble-promotion -Wimplicit-fallthrough -Woverloaded-virtual -Wnon-virtual-dtor -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME scene_cooker COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/scene_cooker" -v)
endif()

option(LUDUS_SCENE_SANITIZERS "Instrument the scene consumer with ASan and UBSan" OFF)
if(LUDUS_SCENE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(reference_scene PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(reference_scene PRIVATE -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
    add_test(NAME scene_capture COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_capture.py" "$<TARGET_FILE:reference_scene>")
    set_tests_properties(scene_capture PROPERTIES TIMEOUT 60)
    add_test(NAME scene_acceptance_driver COMMAND "${Python3_EXECUTABLE}" -m unittest discover
        -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p test_acceptance.py -v)
    add_test(NAME scene_renderer_acceptance COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_acceptance.py" "$<TARGET_FILE:reference_scene>")
    set_tests_properties(scene_renderer_acceptance PROPERTIES TIMEOUT 60)
    option(LUDUS_SCENE_NATIVE_ACCEPTANCE "Require native ray queries in the complete-scene acceptance fixture" OFF)
    if(LUDUS_SCENE_NATIVE_ACCEPTANCE)
        add_test(NAME scene_native_acceptance COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_acceptance.py" "$<TARGET_FILE:reference_scene>" --native)
        set_tests_properties(scene_native_acceptance PROPERTIES TIMEOUT 90)
    endif()
endif()
